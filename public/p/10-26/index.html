#include <stdio.h>
#include <stdlib.h>

int main()
{
	int a, b, t;
	printf("请输入两个数字:\n");
	scanf_s("%d,%d", &a, &b);

	while (a % b != 0)
	{
		t = b;//存上一轮的除数
		b = a % b;//这一轮的余数做下一轮的除数
		a = t;//做下一轮的被除数
	}
	printf("%d", b);//最后剩下的除数就是答案
	return 0;

}

int main(void)
{
    int x, y, gcd;
    printf("请输入两个正整数：");
    scanf("%d%d", &x, &y);
    if(x < y)
        gcd = x;
    else
        gcd = y;
    while(x % gcd ||y % gcd)
        gcd--;
    printf("gcd = %d\n", gcd);

    return 0;
}

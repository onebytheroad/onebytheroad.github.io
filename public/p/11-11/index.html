#include<stdio.h>

// 斐波那契数列函数
int fibonacci(int n) {
    if(n <= 1) {
        return n;
    }
    int a = 0, b = 1;
    for(int i = 2; i <= n; i++) {
        int temp = a + b;
        a = b;
        b = temp;
    }
    return b;
}

int main() {
    int n;
    printf("请输入一个整数：");
    scanf("%d", &n);
    printf("斐波那契数列的第%d项为：%d\n", n, fibonacci(n));
    return 0;
}

#include <stdio.h>

#include <windows.h>

int char_io()
{
	char* str = "hello world";
	char* filepath = "d:\\11\\char_io.txt";

	FILE* fp = NULL;
	errno_t err = fopen_s(&fp, filepath, "w");
	if (err != 0 || fp == NULL)
	{
		return -1;
	}

	while (*str != '\0')
	{
		fputc(*str, fp);
		str++;
	}
	fputc('\0', fp);
	fclose(fp);

	err = fopen_s(&fp, filepath, "r");
	if (err != 0 || fp == NULL)
	{
		return -1;
	}
	while (!feof(fp))//判断是否到达文件末尾
	{
	char ch = fgetc(fp);
	printf("%c", ch);
	}
	printf("\n");
	fclose(fp);

	return 0;
}
int main()
{
	char_io();
	return 0;
}

{


	while (1) 
	{
		record r = { 0 };

		printf("Please input the age:\n");
		scanf_s("%d", &r.age);
		if (r.age == 0)
		{
			break;
		}

		printf("Please input the name:\n");

		fwrite(&r, sizeof(r), 1, fp);
	}

int read_record(FILE* fp)
{
	while (!feof(fp))
	{
		record r = { 0 };
		int count = fread(&r, sizeof(r), 1, fp);
		if (count == 0)
		{
			break;
		}
		printf("age:%d,name:%s\n", r.age, r.name);
	}
	return 0;
}

int main()
{
	FILE* fp = NULL;
	errno_t err = fopen_s(&fp, FILENAME, "w");
	if (err != 0 || fp == NULL)
	{
		return -1;

	}

	write_record(fp);

	fclose(fp);

	err = fopen_s(&fp, FILENAME, "r");
	if (err != 0 || fp == NULL)
	{
		return -1;
	}
	
	read_record(fp);

	fclose(fp);

}

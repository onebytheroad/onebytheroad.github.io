#include <stdio.h>
#include <string.h>
#include <stdlib.h> 
#include <stdbool.h>
#include <malloc.h>
#include <stdint.h>

void print_float_bits(float f) {
    // 1. 将 f 的二进制表示转为 uint32_t
    uint32_t u = *((uint32_t*)&f);

    // 2. 分别提取符号位(1 bit)、指数位(8 bits)、尾数位(23 bits)
    //    并打印每一位
    // 符号位: bit 31
    // 指数位: bit 30~23
    // 尾数位: bit 22~0

    printf("符号位(1): ");
    printf("%d", (u >> 31) & 1);

    printf("\n指数位(8): ");
    for (int i = 30; i >= 23; i--) {
        printf("%d", (u >> i) & 1);
    }

    printf("\n尾数位(23): ");
    for (int i = 22; i >= 0; i--) {
        printf("%d", (u >> i) & 1);
    }
    printf("\n");
}

int main()
{
    print_float_bits(1.0f);

    return 0;
}
